# PTY proof (160x40, isolated session directory)
cd ~/agents/vendor/oh-my-pi
ROOT=/tmp/omp-command-mode-proof-rerun
rm -rf "$ROOT"
mkdir -p "$ROOT/agent" "$ROOT/session" "$ROOT/work"
cp ~/agents/local/proofs/command-mode/command-mode.jsonl "$ROOT/session/command-mode.jsonl"
cp ~/agents/local/proofs/command-mode/rich-mode.jsonl "$ROOT/session/rich-mode.jsonl"
tmux kill-session -t omp-command-mode 2>/dev/null || true
# Start command-mode fixture; press ':' and capture colon-open-160x40.txt.
tmux new-session -d -s omp-command-mode -x 160 -y 40 -c "$PWD" "PI_CODING_AGENT_DIR='$ROOT/agent' bun run dev --resume '$ROOT/session/command-mode.jsonl' --session-dir '$ROOT/session' --cwd '$ROOT/work' --no-title --no-extensions --no-skills --no-rules"
sleep 4
tmux send-keys -t omp-command-mode ':'
sleep 1
tmux capture-pane -p -t omp-command-mode > local/proofs/command-mode/colon-open-160x40.txt
# Restart fixture; capture before, send :wrap, capture after.
tmux kill-session -t omp-command-mode
tmux new-session -d -s omp-command-mode -x 160 -y 40 -c "$PWD" "PI_CODING_AGENT_DIR='$ROOT/agent' bun run dev --resume '$ROOT/session/command-mode.jsonl' --session-dir '$ROOT/session' --cwd '$ROOT/work' --no-title --no-extensions --no-skills --no-rules"
sleep 4
tmux capture-pane -p -t omp-command-mode > local/proofs/command-mode/wrap-before-160x40.txt
tmux send-keys -t omp-command-mode ':wrap' Enter
sleep 1
tmux capture-pane -p -t omp-command-mode > local/proofs/command-mode/wrap-after-160x40.txt
# Restart rich fixture; capture before, send :rich, capture after, then :bogus.
tmux kill-session -t omp-command-mode
tmux new-session -d -s omp-command-mode -x 160 -y 40 -c "$PWD" "PI_CODING_AGENT_DIR='$ROOT/agent' bun run dev --resume '$ROOT/session/rich-mode.jsonl' --session-dir '$ROOT/session' --cwd '$ROOT/work' --no-title --no-extensions --no-skills --no-rules"
sleep 4
tmux capture-pane -p -t omp-command-mode > local/proofs/command-mode/rich-before-160x40.txt
tmux send-keys -t omp-command-mode ':rich' Enter
sleep 1
tmux capture-pane -p -t omp-command-mode > local/proofs/command-mode/rich-after-160x40.txt
tmux send-keys -t omp-command-mode ':bogus' Enter
sleep 1
tmux capture-pane -p -t omp-command-mode > local/proofs/command-mode/bogus-160x40.txt
tmux kill-session -t omp-command-mode
